\chapter{Shared analytic bridges}
\label{chap:shared-analytic-bridges}

The long routes reuse a small collection of shared and branch-specific endpoint interfaces.  This
chapter records them once; it does not make every node a conjunctive premise of every route.
The comparison bridges below are proved in Lean, including smooth/literature
equivalence under intrinsic absolute continuity and the general log-concave Cheeger--Buser
comparison. Interfaces may evolve during formalization without a prior API freeze.
A published input or an upstream reviewed proof receives no \texttt{leanok} merely from its citation.

\section{Three presentations of the Poincar\'e constant}

\begin{theorem}[Smooth tests and ambient relaxation have the same finite constants]
  \label{thm:legacy-relaxed-holds-equivalence}
  \lean{FunctionalInequalities.Poincare.SmoothTest.holds_iff_ambient,
        FunctionalInequalities.Poincare.SmoothTest.holds_iff_preEnergy}
  \leanok
  \uses{lem:preenergy-poincare-relaxation,def:l2-mean-variance,def:smooth-gradient-preenergy,
        def:smooth-gradient-relaxation,def:legacy-smooth-test-poincare}
  Let $\mu$ be a probability law on a finite-dimensional real Hilbert space.  For every finite
  $C\ge0$, the smooth-test inequality on all globally smooth finite-gradient-energy tests holds at $C$
  if and only if the $L^2$ inequality for the ambient sequential relaxation holds at $C$.
  The raw smooth-gradient test family has exactly the same finite-constant predicate.
\end{theorem}
\begin{proof}
  \leanok
  A constant sequence bounds relaxed energy by raw test energy. Conversely, at each
  finite relaxed-energy point use a recovery sequence whose raw energies converge to that value.
  Continuity of $L^2$ variance passes the inequality to the limit for every finite $C\ge0$,
  including $C=0$. Infinite-energy points are outside the finite-energy test domain. No density
  or equality of raw and relaxed energy on individual tests is required.
\end{proof}

\begin{theorem}[Smooth-test and intrinsic smooth-gradient constants agree]
  \label{thm:legacy-canonical-constant-comparison}
  \lean{FunctionalInequalities.Poincare.SmoothTest.constant_eq_intrinsic,
        FunctionalInequalities.Poincare.SmoothTest.constant_eq_ambient,
        FunctionalInequalities.Poincare.SmoothTest.holds_iff_intrinsic}
  \leanok
  \uses{thm:legacy-relaxed-holds-equivalence,
        cor:ambient-intrinsic-constant-equality,def:selected-canonical-frontend}
  For every finite-dimensional Borel probability $\mu$,
  \[
    \CP(\mu)=\CPCan(\mu)
  \]
  as extended nonnegative constants, including the value $+\infty$. Both also equal the ambient
  relaxed optimum; the legacy and selected finite-constant predicates agree for every $C\ge0$.
\end{theorem}
\begin{proof}
  \leanok
   The preceding theorem identifies the sets of valid finite constants for the legacy and
  ambient-relaxed presentations. The relaxed optimizer order law turns these into finite upper
  bounds of an extended constant. If it is finite, its own nonnegative-real value is an allowed
  upper bound; if it is infinite, the set is empty. Their extended optima therefore agree.  The exact
  ambient/intrinsic equality identifies the latter with the selected constant.
\end{proof}

\begin{theorem}[Smooth and locally Lipschitz test constants agree]
  \label{thm:smooth-lipschitz-constant-equivalence}
  \lean{Solutions.ThmSmoothLipschitzConstantEquivalence.certificate}
  \leanok
  \uses{def:affine-support-intrinsic-law,def:literature-poincare-constant,
        def:legacy-smooth-test-poincare,lem:covariance-test-core-density}
  Let $\nu$ be the intrinsic law of a finite-dimensional Borel probability $\mu$.  If $\nu$ is
  absolutely continuous with respect to Lebesgue measure on its intrinsic space, then
  \[
    \CP(\mu)=\CP(\nu)=\CPLit(\mu).
  \]
\end{theorem}
\begin{proof}
  \leanok
  Restriction to the affine support and extension constant in the normal directions give
  the first equality. Every smooth admissible test is locally Lipschitz. Conversely,
  \cref{lem:covariance-test-core-density} with $B=I$ approximates each finite-energy locally
  Lipschitz test in both $L^2$ and gradient $L^2$ by compactly supported smooth tests.
  Variance and energy converge, so each finite smooth-test constant also works on the
  locally Lipschitz domain. The sets of admissible finite constants, and hence their extended
  infima, agree. No raw/core identity for singular ambient laws is used.
\end{proof}

\section{Isoperimetric, affine, and one-dimensional endpoints}

\begin{definition}[Isoperimetric Cheeger constant and inverse scale]
  \label{def:isoperimetric-cheeger-scale}
  \lean{FunctionalInequalities.KLS.Isoperimetry.carrierMeasure,
        FunctionalInequalities.KLS.Isoperimetry.boundaryContent,
        FunctionalInequalities.KLS.Isoperimetry.constant,
        FunctionalInequalities.KLS.Isoperimetry.inverseScale}
  \leanok
  \uses{def:affine-support-intrinsic-law}
  Let $\mu$ be a probability law on a finite-dimensional Euclidean space $E$, and let $A$ be its
  affine support, equipped with the distance induced from $E$.  For $x\in A$ and a nonempty Borel
  set $B\subseteq A$, put
  \[
    d_A(x,B)=\inf_{y\in B}\norm{x-y},
    \qquad
    B^r=\{x\in A:d_A(x,B)<r\},
    \qquad
    \mu^+(B)=\liminf_{r\downarrow0}\frac{\mu(B^r)-\mu(B)}{r}.
  \]
  If $A$ has positive dimension, define the extended nonnegative quantities
  \[
    h_\mu=\inf_{\substack{B\subseteq A\ \mathrm{Borel}\\0<\mu(B)<1}}
      \frac{\mu^+(B)}{\min\{\mu(B),1-\mu(B)\}},
    \qquad
    \PsiKLS(\mu)=h_\mu^{-1},
  \]
  with $0^{-1}=+\infty$ and $(+\infty)^{-1}=0$.  If $A$ is zero-dimensional, set
  $h_\mu=+\infty$ and $\PsiKLS(\mu)=0$.  The isoperimetric number $h_\mu$ is not the metric energy
  $\Ch_\mu$ of \cref{thm:smooth-gradient-relaxation-two-cheeger}.
\end{definition}

\begin{theorem}[Identification with the density Cheeger frontend]
  \label{thm:klartag-density-frontend-identification}
  \uses{def:log-concave-law,thm:borell-logconcave-density,
        def:affine-support-intrinsic-law,def:literature-poincare-constant,
        def:isoperimetric-cheeger-scale,lem:relative-perimeter-cheeger-infimum}
  \lean{Solutions.LemRelativePerimeterCheegerInfimum.intrinsic,
        FunctionalInequalities.Poincare.LocalSlope.Intrinsic.constant_eq_iInf_gradient_of_logConcave}
  \leanok
  Let $\mu$ be a non-Dirac finite-dimensional log-concave probability law, let $A=a+V$ be its
  affine support, and let $\nu$ be its intrinsic law on the positive-dimensional Euclidean space
  $V$.  Let $\rho$ be the upper-semicontinuous log-concave density supplied by
  \cref{thm:borell-logconcave-density}, put $\Omega=\operatorname{int}_V K_\nu$, and define
  \[
    h_\nu^{\mathrm{sm,rel}}
      =\inf_{\substack{U\subseteq V\ \text{smooth domain}\\
                       0<\nu(U)<1}}
        \frac{\displaystyle\int_{\partial_VU\cap\Omega}\rho
          \dd\mathcal H_V^{\dim V-1}}
             {\min\{\nu(U),1-\nu(U)\}},
    \qquad
    \psi_\nu=(h_\nu^{\mathrm{sm,rel}})^{-1},
  \]
  where smooth domains locally lie on one side of their $C^\infty$ boundary, and
  $\mathcal H_V^{\dim V-1}$ is Hausdorff measure for the Euclidean metric on $V$.
  Then
  \[
    h_\mu=h_\nu^{\mathrm{sm,rel}},
    \qquad
    \PsiKLS(\mu)=\psi_\nu,
  \]
  and $\CPLit(\mu)$ equals the infimum of finite $C\ge0$ for which
  \[
    \Var_\nu(g)\le C\int_V\norm{\nabla g}^2\dd\nu
  \]
  holds for every locally Lipschitz $g\in L^2(\nu)$ with
  $\int_V\norm{\nabla g}^2\dd\nu<\infty$, with $\inf\varnothing=+\infty$.  These quantities
  do not depend on the affine origin $a$.
\end{theorem}
\begin{proof}
  \leanok
  Translation to intrinsic coordinates is an isometry. Rademacher differentiability and
  absolute continuity identify the two descriptions of the literature test energy.
  The full proof of the Cheeger equality is \cref{lem:relative-perimeter-cheeger-infimum}:
  distance cutoffs and smooth coarea give one inequality, while outward dilation about an
  interior point followed by convolution gives the other. The latter uses an explicit
  log-concave density bound to avoid a charge on the carrier boundary. The boundary of $U$
  may meet or cross $\partial K_\nu$; no condition $\partial U\subseteq\Omega$ is imposed.
  This proves equality of the two infima, without a testwise perimeter identity. The analytic
  proof is supplied in the appendix. The two isoperimetric clauses $h_\mu=h_\nu^{\mathrm{sm,rel}}$
  and $\PsiKLS(\mu)=\psi_\nu$ are Lean-certified through
  \cref{lem:relative-perimeter-cheeger-infimum}
  (\texttt{Solutions.LemRelativePerimeterCheegerInfimum.intrinsic}, 2026-09-19).  The
  $\CPLit(\mu)$ clause is a local Lean proof as well
  (generic comparison in \texttt{Poincare/LocalSlope/Gradient.lean}, log-concave specialization in
  \texttt{KLS/LogConcave/Poincare.lean}; proved 2026-09-19): $\CPLit$ is defined through the local
  metric slope, and Rademacher differentiability together with the intrinsic absolute continuity of
  every log-concave law identifies the slope with $\norm{\nabla g}$ almost everywhere, hence the
  admissible class and the energy, hence the two infima.  The reference Haar measure is
  constructed inside the proof rather than assumed, so the statement carries neither a reference
  measure nor an affine origin, which is the last sentence of the theorem.  All three clauses being
  local Lean proofs, the node now carries \texttt{leanok}.  The separate
  Cheeger--Buser comparison below is also a local Lean proof as of 2026-09-26;
  the dimension-free KLS bound remains open.
\end{proof}

\begin{theorem}[Cheeger--Buser comparison for log-concave laws]
  \label{thm:logconcave-cheeger-poincare-comparison}
  \uses{def:log-concave-law,def:isoperimetric-cheeger-scale,
        def:literature-poincare-constant,thm:klartag-density-frontend-identification,
        lem:covariance-test-core-density}
  \lean{FunctionalInequalities.KLS.LogConcave.literatureConstant_le_four_mul_inverseScale_sq,
        FunctionalInequalities.KLS.Isoperimetry.smoothTest_constant_le,
        FunctionalInequalities.KLS.LogConcave.inverseScale_sq_le_pi_literature}
  \leanok
  Every finite-dimensional log-concave probability law satisfies
  \[
    \CPLit(\mu)\le4\PsiKLS(\mu)^2,
    \qquad
    \PsiKLS(\mu)^2\le\pi\CPLit(\mu).
  \]
\end{theorem}
\begin{proof}
  \leanok
  If $\mu$ is a Dirac law, then its affine support is a singleton, $\PsiKLS(\mu)=0$, and
  $\CPLit(\mu)=0$ because every test is $\mu$-almost everywhere constant.  Thus both inequalities
  hold directly.  Otherwise apply \cref{thm:klartag-density-frontend-identification}.  Cheeger's
  inequality in the normalization of Milman's Theorem~1.1 and Bobkov's equation~(1.3) gives the
  first inequality; this first inequality is also a local Lean proof
  (\texttt{KLS/Geometry/Cheeger/Inequality.lean}, 2026-09-19: a median split, the layer-cake
  formula, the multiplied Minkowski infimum bound and the Minkowski coarea inequality give
  $h_M^2\int(g-m)_+^2\le4\int_{\{g>m\}}\norm{\nabla g}^2$ for bounded $C^1$ Lipschitz $g$, a
  smooth cutoff extends this to every smooth test, and \cref{lem:relative-perimeter-cheeger-infimum}
  transports it to $\CPLit(\mu)\le4\PsiKLS(\mu)^2$). The second inequality is now also a
  local Lean proof (2026-09-26). The intrinsic Borell law is convolved with the actual normalized
  Gaussian kernel. Its positive smooth convex potential gives the exact smooth-model $\pi$ bound;
  its literature Poincaré constant is at most the source constant plus the Gaussian variance.
  A finite lower bound on the approximants' Minkowski Cheeger constants passes through the
  smooth Lipschitz test inequality under setwise convergence of laws, using weak convergence
  only for bounded continuous gradients. The resulting source-law bound is optimized over
  finite constants and transported through intrinsic coordinates. This includes zero-dimensional
  affine support and zero or infinite optimal constants. The argument does not assert
  convergence of the relative perimeter of each individual domain.
  Klartag's equation~(1.4), with the right-hand bound due to
  De Ponti--Mondino, gives the second; see
  \cite{Milman2009Isoperimetric,Bobkov1999LogConcave,
        Klartag2023Logarithmic,DePontiMondino2021}.
  To make the boundary convention explicit, De Ponti--Mondino define their Cheeger constant
  using metric BV perimeter. Work on the closed intrinsic support $K=K_\nu$, with its induced
  distance and probability measure $\nu$, and write
  \[
    h_{\mathrm{BV}}(K,\nu)
      =\inf_{\substack{B\subseteq K\ \mathrm{Borel}\\0<\nu(B)<1}}
        \frac{\operatorname{Per}_\nu(B)}{\min\{\nu(B),1-\nu(B)\}}.
  \]
  Symmetry of BV perimeter under complementation identifies this with their convention
  $0<\nu(B)\le1/2$. By Lemma~2.1 and Theorem~3.6 of
  \cite{AmbrosioDiMarinoGigli2017Minkowski}, perimeter is the $L^1$ relaxation of outer
  Minkowski content. Hence the corresponding Cheeger infima agree: perimeter is bounded by
  Minkowski content, while recovery sets and continuity of
  $\min\{\nu(B),1-\nu(B)\}$ under $L^1$ convergence give the reverse inequality.
  Restricting a set to $K$ preserves its mass and can only decrease its measured enlargement;
  for sets already in $K$, intrinsic enlargements have the same measure as ambient ones.
  Thus $h_\mu=h_{\mathrm{BV}}(K,\nu)$. This is an equality of infima, not a claim that each
  set's perimeter equals its unrelaxed Minkowski content.

  De Ponti--Mondino's equation~(2) uses the Rayleigh numerator $\int|\nabla f|^2\,d\nu$,
  without the factor $1/2$ in their Cheeger energy. Its tests are nonzero, mean-zero functions
  in $\operatorname{Lip}_b(K)$, meaning bounded Lipschitz functions with bounded support.
  To identify this test class with the literature one, apply
  \cref{lem:covariance-test-core-density} with $B=I$ to a nonzero centered literature test
  $g$, obtaining $g_j\in C_c^\infty(V)$ converging to $g$ in both $L^2(\nu)$ and gradient
  $L^2(\nu;V)$. Fix $\psi\in C_c^\infty(V)$ with $\int\psi\,d\nu=1$ and put
  \[
    u_j=g_j-\Bigl(\int g_j\,d\nu\Bigr)\psi.
  \]
  Since $\int g_j\,d\nu\to0$, the same two convergences hold for $u_j$; its restriction to
  $K$ has zero mean and bounded support, and is nonzero in $L^2$ for all sufficiently large
  $j$. The metric slope on $K$ equals the ambient gradient norm $\nu$-almost everywhere,
  since $\nu(\partial K)=0$. Conversely, a source test has a bounded Lipschitz extension
  to $V$, with the same values and gradient norm $\nu$-almost everywhere. Passing to
  Rayleigh quotients therefore gives $\lambda_1^{-1}=\CPLit(\mu)$, with the extended
  reciprocal conventions. The zero-dimensional case was handled separately above.

  Their Corollary~1.2, equation~(12), gives
  a coefficient strictly smaller than $\pi$ in the zero-curvature case.
  We retain the convenient bound $\pi$ in Klartag's formula and make no optimality
  claim for it. Their sharp Buser estimate concerns the positive-curvature case.
  The numerical inequalities above are local Lean proofs. The metric BV test class
  is the bounded-support \texttt{Lip\_b} class used by De Ponti--Mondino; the
  sequential perimeter definition, actual closed-support law, finite-space complement
  symmetry, and zero-dimensional half-mass test convention have local Lean interfaces
  (2026-09-26). Complement symmetry also holds for every finite measure on every metric
  space, in particular for the closed-support law on an arbitrary unbounded support
  (2026-09-28): the reflection of a test is clamped to $[0,1]$ and cut off radially, and the
  errors are masses outside large balls. Finiteness is necessary, since for Lebesgue measure on
  $\mathbb R$ the empty set has perimeter zero and the line has infinite perimeter. For a
  finite measure, globally Lipschitz approximants with arbitrary constants, values and supports,
  converging to the indicator in the global $L^1(\nu)$ distance, may replace bounded-support
  tests: clamping to $[0,1]$ and a radial cutoff give tests with the same limit and no larger
  liminf cost, so the perimeter is at most that liminf (2026-09-28). These are the approximants
  of Definition~3.3 of \cite{AmbrosioDiMarinoGigli2017Minkowski} (globally Lipschitz, global
  $L^1$ convergence). On a proper metric space the same bound holds for locally Lipschitz
  approximants with global $L^1$ convergence (2026-09-29), a variant discussed in their Remark~3.5
  (not the article's working definition, which is the global $L^1$ class of Definition~3.3): the
  truncation is carried by a compact closed ball, on which a locally Lipschitz function is
  Lipschitz. For every finite Borel measure the relaxation infimum over the approximants of

\begin{theorem}[Perimeter as relaxed Minkowski content]
  \label{thm:relaxed-minkowski-content-perimeter}
  \lean{FunctionalInequalities.KLS.MetricBV.relaxedMinkowskiContent_eq_perimeter_of_finite_on_bounded,
        FunctionalInequalities.KLS.MetricBV.lipschitzRelaxation_eq_perimeter_of_measure_lt_top,
        FunctionalInequalities.KLS.MetricBV.relaxedUpperMinkowskiContent_eq_perimeter_of_finite_on_bounded,
        Solutions.ThmRelaxedMinkowskiContentPerimeter.certificate}
  \leanok
  This is Theorem~3.6 of \cite{AmbrosioDiMarinoGigli2017Minkowski} (arXiv:1603.08412v1), with
  the perimeter of its Definition~3.3.  Let $(X,\mathsf d)$ be a metric space and $\mathfrak m$ a
  Borel measure that is finite on bounded Borel sets.  For a Borel set $A$ with
  $\mathfrak m(A)<\infty$ let
  \[
    \mathcal M_-(A)=\liminf_{r\downarrow 0}\frac{\mathfrak m(A^r)-\mathfrak m(A)}{r},\qquad
    A^r=\{x:\mathsf d_A(x)<r\},
  \]
  and define the relaxed lower Minkowski content and the perimeter by
  \[
    \mathcal M_-^\ast(A)=\inf\Bigl\{\liminf_{h\to\infty}\mathcal M_-(A_h):
      \mathfrak m(A_h\,\triangle\,A)\to 0\Bigr\},\qquad
    \operatorname{Per}(A)=\inf\Bigl\{\liminf_{h\to\infty}\int_X|\nabla f_h|\,d\mathfrak m:
      f_h\in\operatorname{Lip}(X),\ \int_X|f_h-\chi_A|\,d\mathfrak m\to 0\Bigr\},
  \]
  with Borel approximating sets, $|\nabla f|$ the local slope (zero at isolated points) and
  $\inf\emptyset=\infty$.  Then $\mathcal M_-^\ast(A)=\operatorname{Per}(A)$.  The same holds for
  the relaxed upper content, defined with $\limsup$; together these give the second clause of
  Lemma~2.1 of the paper.  These are equalities of infima, not identities between the content
  and the perimeter of an individual set.
\end{theorem}
\begin{proof}
  \leanok
  The relaxed lower and upper contents both equal the bounded-support metric BV perimeter at
  measurable finite-mass targets, under the finiteness of $\mathfrak m$ on bounded Borel sets:
  perimeter is at most the lower content, perimeter is lower semicontinuous under indicator
  $L^1$ convergence, and actual recovery sets have finite mass and contents converging to the
  perimeter.  The truncation and bounded-support reduction of Remark~3.4 identifies the
  Definition~3.3 relaxation with the same perimeter at finite-mass targets, for every Borel
  measure.  The certificate composes the two identities.
\end{proof}

\begin{lemma}[Ambient and intrinsic covariance agree]
  \label{lem:ambient-intrinsic-covariance}
  \lean{FunctionalInequalities.Covariance.operator_apply_eq_zero_of_mem_orthogonal,
        FunctionalInequalities.Covariance.bilin_intrinsic,
        FunctionalInequalities.Covariance.operator_coe_intrinsic,
        FunctionalInequalities.Covariance.opNorm_operator_intrinsic,
        FunctionalInequalities.Covariance.intrinsic_operator_inner_self_pos,
        FunctionalInequalities.Covariance.operator_eq_zero_of_direction_eq_bot,
        Solutions.LemAmbientIntrinsicCovariance.certificate}
  \leanok
  \uses{def:affine-support-intrinsic-law}
  Let $\mu$ be a Borel probability law with finite second moment on a finite-dimensional Euclidean
  space $E$, and let $A=a+V$ be its affine support.  Let $\Cov_{\mu,V}$ be the covariance operator
  of its intrinsic law on $V$.  The ambient covariance $\Cov_\mu$ annihilates $V^\perp$, restricts
  to $\Cov_{\mu,V}$ on $V$, and satisfies
  \[
    \norm{\Cov_\mu}_\op=\norm{\Cov_{\mu,V}}_\op.
  \]
  Moreover, $\Cov_{\mu,V}$ is positive definite on $V$; in dimension zero both covariance
  operators and both norms are zero.
\end{lemma}
\begin{proof}
  \leanok
  The barycenter lies in $A$, because $A$ is closed and convex and carries $\mu$, so the centered
  random vector lies in $V$ almost surely.  Hence the covariance form vanishes whenever one
  argument is in $V^\perp$, and the operator annihilates $V^\perp$.  The intrinsic law is the
  image of $\mu$ under translation by the chosen support point followed by orthogonal projection
  onto $V$; translation leaves the covariance form unchanged and the adjoint of the projection is
  the inclusion, so the intrinsic form is the restriction of the ambient form.  Decomposing an
  ambient vector as its projection plus an orthogonal remainder gives the restriction identity for
  the operators and, with the norm bound of the projection, the equality of operator norms.  If the
  intrinsic form vanished on a nonzero $v\in V$, the variance of $\langle v,\cdot\rangle$ would be
  zero, so $\nu$ would be carried by the proper affine hyperplane on which $\langle v,\cdot\rangle$
  equals its mean, contradicting the fact that the affine support of the intrinsic law is all of
  $V$.  In dimension zero $V^\perp=E$, so both operators are zero.
\end{proof}

  $\nu$ and the mean of $\nu$ lies in $\Omega$, $\Var_{[0,1]}(q)=\Var_\nu(q\circ F)\le\sigma^2$.
  The closed form of $\Var_{[0,1]}(q)$ (through $\int_0^1\log w$ and $\int_0^1\log^2w$) and the
  inequality $ab(\log b-\log a)^2\le(b-a)^2$ give $\Var_{[0,1]}(q)\ge(\sqrt A+\sqrt B)^{-4}$.
  Taking $x_0=x$, perturbing $(A,B)$ to $(A+\varepsilon,B+\varepsilon)$ and letting
  $\varepsilon\to0$ yields $F(x)(1-F(x))\le\sigma\rho(x)$.

  \emph{Sharp Hardy inequality for clipped tests.}  For $a\le b$ in $\Omega$ with
  $F(a)\le\tfrac12\le F(b)$ and a smooth test $f$, let $f_{ab}$ be $f$ clipped to the constants
  $f(a)$ and $f(b)$ outside $[a,b]$, $h=f_{ab}-\int f_{ab}\,d\nu$ and
  $\tilde G(x)=\int_{(-\infty,x]}h\,d\nu$.  Integration by parts gives
  $\int h^2\,d\nu=-\int_a^b f'\tilde G$.  The ground-state identity
  $h^2\rho=\rho\bigl(h-\tilde G(1-2F)/(2W)\bigr)^2+\tilde G^2\rho/(4W^2)+\Xi'$ with
  $\Xi=\tilde G^2(1-2F)/(2W)$, integrated over intervals $[c,e]\subseteq\Omega$ containing
  $[a,b]$ with $F(c)\le\tfrac12\le F(e)$, gives
  $\tfrac14\int_c^e\tilde G^2\rho/W^2\le\int h^2\,d\nu+h(a)^2F(c)+h(b)^2(1-F(e))$ because
  $\tilde G=h(a)F$ left of $a$ and $\tilde G=-h(b)(1-F)$ right of $b$; letting $F(c)\to0$ and
  $F(e)\to1$ gives $\int_a^b\tilde G^2\rho/W^2\le4\int h^2\,d\nu$.  With
  $-f'\tilde G\le\tilde G^2\rho/(8W^2)+2f'^2W^2/\rho$ and the profile bound
  $W^2/\rho\le\sigma^2\rho$ this yields
  $\Var_\nu(f_{ab})=\int h^2\,d\nu\le4\sigma^2\int_a^b f'^2\,d\nu\le4\sigma^2\int f'^2\,d\nu$.

  \emph{Exhaustion.}  The four-point inequality $\rho(a)\rho(e)\le\rho(b)\rho(c)$ for
  $a\le b\le c\le e$, $a+e=b+c$, integrated along translates, gives the monotone hazard rates
  $2c\,(1-F)\le\rho$ on $[m,\infty)$ and $2c\,F\le\rho$ on $(-\infty,m]$.  For an admissible test
  $f$ and $a\le m\le b$ in $\Omega$, integration by parts against $F$ (resp.\ $1-F$), the
  elementary bound $2xy\le c\,x^2+y^2/c$ and a directed union over rational endpoints give the
  anchored half-line bounds $\int_{x\le a}(f-f(a))^2\,d\nu\le c^{-2}\int_{x\le a}f'^2\,d\nu$ and
  $\int_{x>b}(f-f(b))^2\,d\nu\le c^{-2}\int_{x>b}f'^2\,d\nu$, which can be made arbitrarily
  small by absolute continuity of $\int f'^2\,d\nu$, choosing $a,b$ with $F(a)$ and $1-F(b)$
  small.  Since $\int(f-f_{ab})^2\,d\nu$ is bounded by the sum of the two tails and
  $(u+v)^2\le(1+\delta)u^2+(1+\delta^{-1})v^2$ for $\delta>0$,
  $\Var_\nu(f)\le(1+\delta)\Var_\nu(f_{ab})+(1+\delta^{-1})\int(f-f_{ab})^2\,d\nu$; letting the
  tails and then $\delta$ tend to zero yields $\Var_\nu(f)\le4\sigma^2\int f'^2\,d\nu$.  The
  smooth-test optimum equals the literature optimum for log-concave laws
  (\cref{thm:selected-literature-constant-equivalence}).

  \emph{The constant $12$, independently.}  With $a=b=m$ the anchored half-line bounds give
  $\Var_\nu(f)\le\int (f-f(m))^2\,d\nu\le c^{-2}\int (f')^2\,d\nu$.  A supporting exponential
  $\rho(s)\le c\,e^{k(s-m)}$ at the median gives $\rho\le c+k(F-\tfrac12)$ on $\Omega$; hence
  $\varphi(F(x))\le x-m$ for $x\ge m$ and $\varphi(F(x))\ge x-m$ for $x\le m$ with
  $\varphi(u)=(u-\tfrac12)/c-k(u-\tfrac12)^2/(2c^2)$.  Since $F(X)$ is uniform on $[0,1]$
  under $\nu$, $\int (x-m)(F(x)-\tfrac12)\,d\nu\ge\int_0^1\varphi(u)(u-\tfrac12)\,du
  =1/(12c)$, and $ab\le c\,a^2/2+b^2/(2c)$ with $\int (F-\tfrac12)^2\,d\nu=1/12$ yields
  $\Var_\nu\ge1/(12c^2)$, so $c^{-2}\le12\Var_\nu$.
\end{proof}

\noindent\textbf{Remark (sharpness, not Lean-certified here).}
The constant $4$ is optimal: the exponential law has $\CPLit=4\Var$, so no smaller universal
coefficient is possible.  The certificate above proves the upper bound only; the exponential-law
equality and the resulting optimality are not formalized in this repository.  The constant-$12$
route uses two estimates that are sharp for the exponential and uniform laws respectively;
these sharpness assertions are also informal background.

The preceding nodes are deliberately modest.  They do not provide a self-adjoint generator, identify a
form gap with the reciprocal of an operator eigenvalue, or prove any of the four long-route gates.

\begin{theorem}[Bakry--\'Emery Poincar\'e bound]
  \label{thm:bakry-emery-poincare}
  \lean{FunctionalInequalities.KLS.LogConcave.literatureConstant_le_inv_of_eq_tilted,
        FunctionalInequalities.KLS.LogConcave.smoothTest_holds_inv_of_eq_tilted,
        FunctionalInequalities.KLS.LogConcave.norm_center_sq_le_inv_mul,
        Solutions.ThmBakryEmeryPoincare.certificate}
  \leanok
  \uses{def:log-concave-law,def:literature-poincare-constant}
  This is Corollary~4.8.2 of \cite{BakryGentilLedoux2014}, in a reviewed stronger form.  Let $V$
  be a finite-dimensional real inner product space with Lebesgue measure, $W:V\to\R$ of class
  $C^2$ with $\nabla^2W(x)(v,v)\ge\rho\norm v^2$ for all $x,v$ and some $\rho>0$, and
  $\nu=e^{-W}dx/\int e^{-W}$ a probability measure.  Then
  \[
    \CPLit(\nu)\le\rho^{-1},
  \]
  and $\Var_\nu(f)\le\rho^{-1}\int\norm{\nabla f}^2\,d\nu$ for every smooth $f\in L^2(\nu)$ with
  $\norm{\nabla f}\in L^2(\nu)$.  The book assumes $V=\R^n$, $W$ smooth and $\int e^{-W}=1$.
  The certificate also constructs the scaled Gaussian $e^{-\rho\norm x^2/2}dx/Z$ as a probability
  measure and applies the bound to it, so the hypotheses are satisfiable in every dimension,
  including dimension zero.  Sharpness of $\rho^{-1}$ and the general Markov-triple
  $CD(\rho,\infty)$ statement (Proposition~4.8.1) are not claimed.
\end{theorem}
\begin{proof}
  \leanok
  The proof follows the book's semigroup route in the Euclidean heat model with generator
  $-\bar T^\dagger\bar T$, where $\bar T$ is the closure of the weighted gradient on $C_c^\infty$.
  The Hessian bound gives $\Gamma_2\ge\rho\Gamma$ and the finite-time decay
  $\Var(u)-\Var(P_tu)\le\rho^{-1}(1-e^{-2\rho t})\norm{\bar Tu}^2$.  Ergodicity follows from
  $\ker\bar T=$ constants (weak gradient and translation invariance) and strong convergence of
  the semigroup on $(\ker\bar T)^\perp$, so $\Var(P_tu)\to0$ and $\Var(u)\le\rho^{-1}\norm{\bar Tu}^2$
  on the form domain.  The optimal constant on the form domain equals the literature constant.
  Log-concavity and full-dimensional support of $\nu$ are derived from the hypotheses.
\end{proof}

\begin{lemma}[Partial symmetrization of tensors]
