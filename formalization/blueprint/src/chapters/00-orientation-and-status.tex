\chapter{Orientation and status}

The mathematical source for the route layer is the public repository
\sourcerepository{} at commit \sourcecommit.  The source is pinned so that a later edit of
the manuscript cannot silently change this graph.  The classical conjecture goes back to
Kannan, Lov\'asz and Simonovits \cite{KannanLovaszSimonovits1995}.  For a modern survey see
Lee and Vempala \cite{LeeVempala2018}.  Eldan \cite{Eldan2013ThinShell} and the later Annals

and quantitative progress toward KLS, not surveys and not proofs of the full conjecture.

This Blueprint is not a transcription of that manuscript.  It exposes the interfaces that the
Lean project needs, and it records non-claims at every boundary where two commonly equivalent
mathematical presentations have not yet been identified formally.

\section{The target}

\begin{definition}[Log-concave probability law]
  \label{def:log-concave-law}
  \lean{FunctionalInequalities.KLS.LogConcave}
  \leanok
  A Borel probability law $\mu$ on a finite-dimensional real normed vector space is log-concave
  when,
  for all compact sets $K_0,K_1$ and every $\lambda\in(0,1)$,
  \[
    \mu\bigl(\lambda K_0+(1-\lambda)K_1\bigr)
      \ge \mu(K_0)^\lambda\mu(K_1)^{1-\lambda}.
  \]
  The compact-set formulation ensures that the Minkowski combination is Borel and includes laws
  carried by a proper affine subspace.
\end{definition}

\begin{theorem}[Compact-set and Borell formulations agree]
  \label{thm:compact-borell-logconcavity-equivalence}
  \lean{FunctionalInequalities.KLS.innerMeasure,
        FunctionalInequalities.KLS.BorelLogConcave,
        FunctionalInequalities.KLS.LogConcave.borel_iff,
        Solutions.ThmCompactBorellLogconcavityEquivalence.certificate}
  \leanok
  \uses{def:log-concave-law}
  For an arbitrary set $S$, write
  \[
    \mu_*(S)=\sup\{\mu(K):K\subseteq S,\ K\text{ compact}\}.
  \]
  A finite-dimensional Borel probability law is log-concave in the sense of
  \cref{def:log-concave-law} if and only if, for all Borel sets $B_0,B_1$ and every
  $\lambda\in(0,1)$,
  \[
    \mu_*\bigl(\lambda B_0+(1-\lambda)B_1\bigr)
      \ge \mu(B_0)^\lambda\mu(B_1)^{1-\lambda}.
  \]
\end{theorem}
\begin{proof}
  \leanok
  Compact sets are Borel and $\mu_*(S)\le\mu(S)$ for every set $S$, so the Borel formulation
  specializes to the compact-set definition.  Conversely, finite-dimensional Borel probability
  laws are inner regular: $\mu(B_i)=\sup\{\mu(K_i):K_i\subseteq B_i\text{ compact}\}$.
  Positive powers and products commute with these suprema in $[0,\infty]$, and for compact
  $K_i\subseteq B_i$ the combination $\lambda K_0+(1-\lambda)K_1$ is a compact subset of
  $\lambda B_0+(1-\lambda)B_1$, being the continuous image of $K_0\times K_1$.  The compact-set
  inequality on each pair $(K_0,K_1)$ therefore bounds $\mu(K_0)^\lambda\mu(K_1)^{1-\lambda}$ by
  $\mu_*(\lambda B_0+(1-\lambda)B_1)$, and the two suprema give the claim.  Null sets are handled
  by the empty compact subset.  This is the precise bridge to the measure-theoretic formulation
  used in the cited literature; it is not implemented by evaluating a measure as an outer measure
  on a possibly non-Borel Minkowski combination.  Compare Bobkov's equation~(5.1) in
  \cite{Bobkov1999LogConcave}.
\end{proof}

\begin{lemma}[Intrinsic absolute continuity of log-concave laws]
  \label{lem:logconcave-intrinsic-absolute-continuity}
  \lean{Solutions.LemLogConcaveIntrinsicAbsoluteContinuity.certificate}
  \leanok
  \uses{def:log-concave-law,def:affine-support-intrinsic-law}
  Let $\nu$ be the intrinsic law of a finite-dimensional log-concave probability $\mu$ on the
  direction space $V$ of its affine support. For every additive Haar measure $m$ on $V$,
  $\nu\ll m$. This includes singular ambient laws and zero-dimensional affine supports.
\end{lemma}
\begin{proof}
  \leanok
  Compact-set log-concavity passes through affine-isometric reconstruction and makes the
  intrinsic support convex. Its affine span is $V$, so its interior is nonempty. Put
  $q_r(x)=\nu(\overline B(x,r))$ and $b_r=m(\overline B(0,r))$.
  Compactness supplies a uniform positive mass bound for fixed-radius balls with centers
  in one interior compact ball. Tonelli gives $\int q_r\,\dd m=b_r$ and a localized lower
  bound, forcing points with $q_r\ge c b_r$ near every such center, uniformly for small $r$.
  Superlevel convexity and separation then give $q_r\ge c b_r$ throughout a fixed ball
  $B(a,R)$. For any $x$, the midpoint inequality bounds $q_r$ below by
  $\sqrt{q_r(x)c b_r}$ on $B((x+a)/2,R/2)$. Integrating and squaring yields
  $q_r(x)\le b_r/(c\,m(B(0,R/2))^2)$. Besicovitch covering converts these uniformly
  bounded small-ball ratios into measure domination and hence absolute continuity.
  This proves the absolute-continuity portion of \cite{Borell1974Convex} directly from the
  compact-set convention; it does not assume a density representative or the Borel-set bridge.
\end{proof}

\begin{theorem}[Borell representation on the affine support]
  \label{thm:borell-logconcave-density}
  \lean{FunctionalInequalities.KLS.IsBorellDensity,
        FunctionalInequalities.KLS.LogConcave.exists_isBorellDensity,
        FunctionalInequalities.KLS.IsBorellDensity.pos_of_mem_interior_support,
        FunctionalInequalities.KLS.IsBorellDensity.exists_bounds_of_isCompact,
        FunctionalInequalities.KLS.LogConcave.closure_convexHull_intrinsic_support,
        FunctionalInequalities.KLS.LogConcave.intrinsic_interior_support_compl,
        FunctionalInequalities.KLS.LogConcave.intrinsic_integrable_exp_norm,
        FunctionalInequalities.KLS.LogConcave.intrinsic_integrable_norm_pow,
        Solutions.ThmBorellLogconcaveDensity.certificate}
  \leanok
  \uses{def:log-concave-law,thm:compact-borell-logconcavity-equivalence,
        def:affine-support-intrinsic-law}
  Let $\nu$ be the intrinsic law of a finite-dimensional log-concave probability $\mu$ on the
  direction space $V$ of its affine support.  Let $K_\nu$ be the closed convex hull of the
  topological support of $\nu$, and put $\Omega=\operatorname{int}_V(K_\nu)$.  There is an
  upper-semicontinuous log-concave Borel function $\rho:V\to[0,\infty)$ such that
  \[
    \nu(V\setminus\Omega)=0,
    \qquad
    \dd\nu=\rho\dd\mathrm{Leb}_V.
  \]
  Moreover, $\rho>0$ on $\Omega$ and
  $\rho,\rho^{-1}\in L^\infty_{\mathrm{loc}}(\Omega,\mathrm{Leb}_V)$, and $\nu$ has finite moments
  of every order.  In dimension zero, Lebesgue measure means counting measure on the one-point
  space, so the assertion includes Dirac laws.
\end{theorem}
\begin{proof}
  \leanok
  The absolute-continuity input is \cref{lem:logconcave-intrinsic-absolute-continuity}, and the
  support of $\nu$ is closed and convex, so $K_\nu=\operatorname{supp}\nu$ and
  $\Omega=\operatorname{int}\operatorname{supp}\nu$; the frontier of a convex set is Lebesgue-null,
  which gives $\nu(V\setminus\Omega)=0$.  For the representative, put $r_n=1/(n+1)$ and
  $\varphi_n(x)=\nu(\overline B(x,r_n))/\mathrm{Leb}(\overline B(0,r_n))$.  Equal-radius ball masses
  satisfy the log-concavity inequality, so each $\varphi_n$ is log-concave, hence so is
  $\psi=\liminf_n\varphi_n$; the proved small-ball estimate bounds $\psi$ uniformly, and the
  Besicovitch differentiation theorem identifies $\psi$ with the Radon--Nikodym derivative
  almost everywhere.  Let $\rho$ be the upper semicontinuous envelope
  $\rho(x)=\inf_n\sup_{B(x,r_n)}\psi$.  It is upper semicontinuous, log-concave, finite, and
  agrees with $\psi$ off the frontier of the convex set $\{\psi>0\}$: outside its closure both
  vanish, and on its interior $\log\psi$ is a finite concave function, hence continuous.  Thus
  $\dd\nu=\rho\,\dd\mathrm{Leb}_V$.  For any such representative, $\{\rho>0\}$ is convex; if it
  had empty interior it would be Lebesgue-null and could not carry $\nu$, so
  $\operatorname{supp}\nu\subseteq\overline{\{\rho>0\}}$ gives
  $\Omega\subseteq\operatorname{int}\{\rho>0\}$ and $\rho>0$ on $\Omega$.  On a compact
  $K\subset\Omega$, upper semicontinuity bounds $\rho$ above and continuity of the concave
  function $\log\rho$ bounds it below by a positive constant; these pointwise bounds imply the
  $L^\infty_{\mathrm{loc}}$ statement.  For the moments, choose a ball $B(0,R)$ of mass at least
  $3/4$ by exhaustion; Borell's lemma, applied through the Borel-set formulation of
  \cref{thm:compact-borell-logconcavity-equivalence} with $t=2/(k+1)$, gives
  $\nu(\|v\|\ge kR)\le 3^{(1-k)/2}$, so $e^{t\|v\|}$ with $t=\log 3/(4R)$ is dominated by a
  geometric series and all moments are finite.  This is imported finite-dimensional
  convex-measure theory in the sense of \cite{Borell1974Convex}, now proved locally from the
  compact-set convention; the reference measure is any additive Haar measure, which in dimension
  zero is a positive multiple of the Dirac mass.
\end{proof}

\begin{definition}[Intrinsic local-slope Poincar\'e constant]
  \label{def:literature-poincare-constant}
  \lean{FunctionalInequalities.Energy.LocalSlope.apply,
        FunctionalInequalities.Energy.LocalSlope.energy,
        FunctionalInequalities.Energy.LocalSlope.Admissible,
        FunctionalInequalities.Poincare.LocalSlope.Holds,
        FunctionalInequalities.Poincare.LocalSlope.Intrinsic.constant}
  \leanok
  Let $\mu$ be a Borel probability measure on a finite-dimensional Euclidean space, and let $A$
  be its affine support.  For a locally Lipschitz function on $A$, write $\abs{D_Af}$ for its local
  metric slope.  A finite $C\ge0$ is valid when
  \[
    \Var_\mu(f)\le C\int \abs{D_Af}^{2}\dd\mu
  \]
  for every locally Lipschitz $f\in L^2(\mu)$ with
  $\int\abs{D_Af}^2\dd\mu<\infty$.  Define $\CPLit(\mu)\in[0,\infty]$ as the infimum of all
  valid finite constants, with $\inf\varnothing=+\infty$.  The test domain is independent of
  $C$; zero-energy tests are included.  For log-concave laws,
  the slope agrees almost everywhere with the tangential gradient on the relative interior of the
  convex support within $A$.  This convention also makes the definition meaningful for singular
  Borel measures.
\end{definition}

\begin{conjecture}[KLS, affine and isotropic forms]
  \label{conj:kls}
  \lean{FunctionalInequalities.KLS.Conjecture,
        FunctionalInequalities.KLS.kls}
  \leanok
  \uses{def:log-concave-law,def:literature-poincare-constant}
  There is a universal $C<\infty$ such that, on every finite-dimensional Euclidean space, every
  log-concave Borel probability $\mu$ with finite second moment satisfies
  \[
    \CPLit(\mu)\le C\norm{\Cov_\mu}_{\op}.
  \]
\end{conjecture}
\begin{proof}
  \leanok
  \uses{thm:bkl-poincare, thm:affine-isotropic-reduction}
  BKL Theorem~1.1 (Theorem~\ref{thm:bkl-poincare}) gives one universal $C$ with $\CPLit(\mu)\le C$ for every isotropic
  log-concave law in every finite-dimensional space. The affine-support and whitening bridge
  (Theorem~\ref{thm:affine-isotropic-reduction}) transfers this bound, with the same constant, to every log-concave law
  with finite second moment, including singular and lower-dimensional supports and dimension~$0$. In Lean,
  \texttt{KLS.kls} is \texttt{KLS.BKL.Localization.conjecture}.
\end{proof}

The equivalence with the isotropic formulation is not hidden in the word ``equivalently.''  It is
the named affine-support and whitening bridge \cref{thm:affine-isotropic-reduction}.

The statement keeps its stable label and environment, and it is proved here.
The Lean name \texttt{FunctionalInequalities.KLS.Conjecture} is a \texttt{def \ldots : Prop}: it names the sentence.
The theorem \texttt{FunctionalInequalities.KLS.kls} in \texttt{KLS/Target.lean} proves it through the BKL route, on the
owner-selected cutoff localization. Its axiom audit uses the standard axioms only, and no local declaration is excluded from the
declaration audit. The route evidence of this node is local Lean evidence; the pinned upstream record is not promoted.
The four exploratory route records remain separate and open; this proof does not complete them.

\section{Status contract}

Public Lean interfaces may evolve during this exploratory stage, including those of proved
results. Proof work may refine a draft without a prior freeze. Certification records the current
statement and proof; it makes no promise of signature stability. Changes to mathematical
hypotheses, conclusions, test domains, or normalization must be recorded and checked against
the original goal, with the affected proofs and consumers updated.

The graph uses the following meanings.

\begin{itemize}
  \item \texttt{\textbackslash lean\{name\}} maps a Blueprint node to an existing declaration.
  \item \texttt{\textbackslash leanok} in both a statement and its proof records a checked Lean
        statement and proof.
  \item Absence of \texttt{leanok} means unfinished formalization, even when the mathematical
        argument has been reviewed elsewhere.
  \item \texttt{\textbackslash notready} marks an explicit unresolved statement choice, such as
        its test domain, hypotheses, or normalization. The nearby text identifies that choice.
        An open proof or possible future API changes alone do not justify the marker.
\end{itemize}

Removing \texttt{notready} does not add \texttt{leanok} or change a route's mathematical status.
Historical freeze records remain dated review snapshots. The current development path permits
\texttt{drafted} to become \texttt{proved} directly after certification of the current version.

There are four levels, and their names are not interchangeable.  The energy-parametrized optimizer
of \cref{chap:canonical-basic} accepts any extended energy.  The selected canonical frontend of
\cref{chap:smooth-affine} chooses the intrinsic-affine smooth-gradient relaxation.  The accepted
smooth-test frontend of \cref{chap:certified-models} quantifies over globally smooth finite-energy
tests, while \cref{def:literature-poincare-constant} quantifies over locally Lipschitz tests.  The
first two are Lean definitions. The selected and legacy constants are now proved equal in
finite dimension. Their comparison with the locally Lipschitz literature constant has an
analytic approximation proof in \cref{chap:analytic-comparison-proofs}, as do the relative
perimeter and covariance-core identifications. These statements still await Lean formalization
and certification. The active conditional-fiber route uses a general finite input
$\kappa$; its structural form lemma (\cref{lem:conditional-fiber-form}), its parameterized
comparison with the gradient energy (\cref{lem:conditional-fiber-energy-comparison}) and the
log-concave specialization that supplies the input $\kappa=4$
(\cref{cor:conditional-fiber-twelve-bound}) are Lean-certified.  The published choice
$\kappa=12$ is the immediate weakening of $4$.  The remaining gate of the route is the
test-independent frame question \cref{q:conditional-fiber-frame}.
